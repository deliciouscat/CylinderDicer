# 개요
locale 문자열 조회 모듈. 화면 문자열 하드코딩 방지. `assets/locale/<locale>.json`에서 key를 읽고 parameter 치환한다.

# 의존성
- `assets/locale/ko.json`
- `assets/locale/en.json`
- `assets/locale/ja.json`
- `game/model/selectors.lua`: hint key 제공.
- `ui/*`: label, hint, result text 조회.

# I/O
- 입력:
  - `set_locale(locale)`.
  - `t(key, params)`.
- 출력:
  - locale 문자열.
  - 누락 key fallback 문자열: `[[missing.key]]`.

# 의사코드
```lua
-- Pattern: Facade + lazy load + 파라미터 보간. 화면 문자열 접근 단일 창구.
local M = {}
local cache   = {}      -- locale -> { key -> template }
local current = "ko"

local function load_locale(locale)
    -- assets/locale/<locale>.json 로드 후 cache에 보관 (1회)
    return decode_json(sys.load_resource("/assets/locale/" .. locale .. ".json"))
end

function M.set_locale(locale)
    current = locale
    cache[locale] = cache[locale] or load_locale(locale)
end

function M.t(key, params)
    local tmpl = (cache[current] or {})[key]
    if not tmpl then return "[[" .. key .. "]]" end        -- 누락 키 가시화(개발용 fallback)
    return (tmpl:gsub("{(%w+)}", function(name)            -- {count} 같은 토큰 치환
        return tostring((params or {})[name])
    end))
end

return M
```
